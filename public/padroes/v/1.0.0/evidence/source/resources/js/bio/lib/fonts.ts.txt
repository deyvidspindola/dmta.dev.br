 * Fontes que o dono da bio pode escolher (Google Fonts, carregadas sob demanda).
 * Inter já vem no HTML da bio — é o padrão quando nada foi escolhido.

  id: string

  /** Agrupamento no seletor do editor. */
  category: BioFontCategory
}

/** Fontes oferecidas para o texto corrido (legíveis em tamanho pequeno). */
export const BODY_FONT_IDS = ['inter', 'dm-sans', 'manrope', 'jakarta', 'outfit', 'sora', 'fraunces']

const CATEGORY_ORDER: BioFontCategory[] = ['sans', 'serif', 'display', 'script']

/** Fontes agrupadas para `<optgroup>` nos seletores do editor. */
export function bioFontsByCategory(): Array<{ category: BioFontCategory; label: string; fonts: BioFont[] }> {
  return CATEGORY_ORDER.map((category) => ({
    category,
    label: BIO_FONT_CATEGORY_LABELS[category],
    fonts: BIO_FONTS.filter((font) => font.category === category),
  })).filter((group) => group.fonts.length > 0)
}

export function resolveBioFont(id?: string): BioFont | undefined {
  if (!id) return undefined
  return BIO_FONTS.find((font) => font.id === id)
}

const loaded = new Set<string>()

/** Injeta um <link> do Google Fonts por fonte ainda não carregada nesta página. */
export function loadBioFonts(ids: Array<string | undefined>): void {
  if (typeof document === 'undefined') return
  for (const id of ids) {
    const font = resolveBioFont(id)
    if (!font?.google || loaded.has(font.id)) continue
    loaded.add(font.id)
    const link = document.createElement('link')
    link.rel = 'stylesheet'
    link.href = `https://fonts.googleapis.com/css2?family=${font.google}&display=swap`
    link.dataset.bioFont = font.id
    document.head.appendChild(link)
  }
}

/** Reseta o cache de fontes injetadas — use em afterEach de testes de browser/jsdom. */
export function resetLoadedBioFonts(): void {
  loaded.clear()
  if (typeof document === 'undefined') return
  document.head.querySelectorAll('link[data-bio-font]').forEach(node => node.remove())
}
