import { LogOut, Plus, Moon, Sun, ArrowUpRight } from 'lucide-react'
import { useState } from 'react'
import { BrandWordmark } from '../../shared/brand/BrandWordmark'
import { csrfToken } from '../../shared/csrf'
import { applyTheme, getStoredTheme } from '../../editor/lib/theme'

export function AdminTopbar({ onNewBio }: { onNewBio: () => void }) {
  const [theme, setTheme] = useState(getStoredTheme)
  return (
    <header className="admin-header">
      <a href="/admin" className="admin-brand" aria-label="Links na Bio — Administração"><BrandWordmark withIcon /><span className="admin-brand__caption">Administração</span></a>
      <a href="/app" className="btn-outline admin-icon admin-editor-link" title="Abrir meu editor" aria-label="Abrir meu editor"><ArrowUpRight size={18} /></a>
      <button type="button" className="btn-outline admin-icon" aria-label={theme === 'dark' ? 'Ativar modo claro' : 'Ativar modo escuro'} onClick={() => { const next = theme === 'dark' ? 'light' : 'dark'; applyTheme(next); setTheme(next) }}>{theme === 'dark' ? <Sun size={18} /> : <Moon size={18} />}</button>
      <button type="button" className="admin-header-action" onClick={onNewBio}><Plus size={18} /><span className="hidden sm:inline">Novo cliente</span><span className="sr-only sm:hidden">Novo cliente</span></button>
      <form method="POST" action="/logout"><input type="hidden" name="_token" value={csrfToken()} /><button type="submit" className="btn-outline admin-icon" title="Sair" aria-label="Sair"><LogOut size={18} /></button></form>
    </header>
  )
}
