<!DOCTYPE html>
<html lang="pt-BR">
    <head>
        <meta charset="utf-8">
        <meta name="viewport" content="width=device-width, initial-scale=1">
        <title>{{ $title }}</title>
        <meta name="description" content="{{ $description }}">
        @php
            $brandMeta = is_array($config['brand'] ?? null) ? $config['brand'] : [];
            $faviconPath = trim((string) ($brandMeta['favicon'] ?? ''));
            if ($faviconPath === '') {
                $faviconPath = trim((string) ($brandMeta['logo'] ?? ''));
            }
            if ($faviconPath !== '') {
                $faviconHref = str_starts_with($faviconPath, 'http')
                    ? $faviconPath
                    : asset(ltrim($faviconPath, '/'));
                $faviconLower = strtolower($faviconPath);
                $faviconType = str_ends_with($faviconLower, '.svg')
                    ? 'image/svg+xml'
                    : (str_ends_with($faviconLower, '.png') ? 'image/png' : 'image/png');
            }
        @endphp
        @if (! empty($faviconHref ?? null))
            <link rel="icon" type="{{ $faviconType }}" href="{{ $faviconHref }}">
        @else
            <link rel="icon" type="image/svg+xml" href="/favicon.svg">
        @endif
        <link rel="preconnect" href="https://fonts.googleapis.com">
        <link rel="preconnect" href="https://fonts.gstatic.com" crossorigin>
        <link href="https://fonts.googleapis.com/css2?family=Inter:wght@400;500;600;700&display=swap" rel="stylesheet">
        @viteReactRefresh
        @vite('resources/js/bio/main.tsx')
    </head>
    <body>
        <div id="root"></div>
        <script>
            window.__BIO_CONFIG__ = @json($config);
            window.__ANALYTICS_KEY__ = @json($bio->analytics_key);
            window.__BIO_SLUG__ = @json($bio->slug ?? null);
            window.__BIO_HOME_HREF__ = @json(request()->attributes->get('tenant_bio') ? '/' : (isset($bio->slug) ? '/'.$bio->slug : '/'));
            window.__BIO_PAGE_PREFIX__ = @json(isset($bio->slug) ? '/'.$bio->slug : '');
        </script>
