import { test, expect } from './fixtures'

for (const width of [390, 1440]) test(`select compartilhado no editor em ${width}px salva a escolha`, async ({ page }) => {
  await page.setViewportSize({ width, height: 900 })
  await page.goto('/login')
  await page.getByLabel('E-mail', { exact: true }).fill('browser-6@example.test')
  await page.getByPlaceholder('Digite sua senha', { exact: true }).fill('Browser-test-only-42')
  await page.getByRole('button', { name: 'Entrar', exact: true }).click()
  await expect(page).toHaveURL(/\/app$/)
  if (width < 768) await page.getByRole('button', { name: 'Mais', exact: true }).click()
  await page.getByRole('button', { name: 'Perfil', exact: true }).click()
  await page.getByRole('tab', { name: 'Textos', exact: true }).click()
  const advanced = page.getByText('Conhecer opções avançadas', { exact: true })
  if (await advanced.isVisible()) await advanced.click()
  const select = page.getByRole('combobox', { name: 'Fonte do nome', exact: true })
  await select.click()
  const option = page.getByRole('option').filter({ hasText: 'Outfit' }).first()
  await option.click()
  await expect(select).toContainText('Outfit')
  if (await page.getByRole('button', { name: 'Salvar', exact: true }).isVisible()) await page.getByRole('button', { name: 'Salvar', exact: true }).click()
  await expect.poll(async () => (await (await page.request.get('/api/bio/load')).json()).config.brand.theme.nameFont).toBe('outfit')
  await expect(page.locator('select:visible')).toHaveCount(0)
  await page.screenshot({ path: `/tmp/shared-fields-app-${width}.png` })
})

test('calendário do editor mantém data ao digitar horário e salva no fuso correto', async ({ page }) => {
  await page.goto('/login')
  await page.getByLabel('E-mail', { exact: true }).fill('browser-6@example.test')
  await page.getByPlaceholder('Digite sua senha', { exact: true }).fill('Browser-test-only-42')
  await page.getByRole('button', { name: 'Entrar', exact: true }).click()
  await expect(page).toHaveURL(/\/app$/)
  await page.getByRole('button', { name: 'Minha página', exact: true }).click()
  await page.frameLocator('iframe').first().locator('[data-preview-item]').first().click()
  await page.getByRole('switch', { name: 'Agendar exibição deste card' }).click()
  await page.getByRole('button', { name: 'Escolher data e hora: Aparecer a partir de', exact: true }).click()
  const calendar = page.getByRole('dialog', { name: 'Calendário: Aparecer a partir de', exact: true })
  const months = await page.evaluate(() => (2027 - new Date().getFullYear()) * 12 + 11 - new Date().getMonth())
  for (let index = 0; index < Math.abs(months); index++) await calendar.getByRole('button', { name: months >= 0 ? 'Próximo mês' : 'Mês anterior', exact: true }).click()
  await calendar.getByRole('button', { name: '25 de dezembro de 2027', exact: true }).click()
  await calendar.getByText('Digitar horário', { exact: true }).click()
  const time = page.getByRole('textbox', { name: 'Horário: Aparecer a partir de', exact: true })
  await time.fill('09')
  await expect(calendar.getByRole('button', { name: '25 de dezembro de 2027', exact: true })).toHaveAttribute('aria-pressed', 'true')
  await time.fill('')
  await time.pressSequentially('093')
  await expect(time).toHaveValue('09:3')
  await time.pressSequentially('0')
  await expect(time).toHaveValue('09:30')
  await time.fill('29:70')
  await expect(time).toHaveValue('09:30')
  await time.fill('0930')
  await expect(time).toHaveValue('09:30')
  await calendar.getByRole('button', { name: 'Aplicar', exact: true }).click()
  if (await page.getByRole('button', { name: 'Salvar', exact: true }).isVisible()) await page.getByRole('button', { name: 'Salvar', exact: true }).click()
  await expect.poll(async () => {
    const data = await (await page.request.get('/api/bio/load')).json()
    return data.config.sections.flatMap((section: { items: Array<{ title: string; schedule?: { from?: string } }> }) => section.items).find((item: { schedule?: { from?: string } }) => item.schedule?.from === '2027-12-25T09:30:00-03:00')?.schedule?.from
  }).toBe('2027-12-25T09:30:00-03:00')
})
