export type Theme = 'dark' | 'light'

const STORAGE_KEY = 'lnb-editor-theme'

export function getStoredTheme(): Theme {
  try {
    const stored = localStorage.getItem(STORAGE_KEY) ?? localStorage.getItem('insta-bio-admin-theme')
    if (stored === 'light' || stored === 'dark') return stored
  } catch { /* O editor continua funcionando sem armazenamento local. */ }
  return 'light'
}

export function applyTheme(theme: Theme) {
  document.documentElement.setAttribute('data-theme', theme)
  try { localStorage.setItem(STORAGE_KEY, theme) } catch { /* Preferência válida durante a sessão. */ }
}

export function initTheme() {
  applyTheme(getStoredTheme())
}
